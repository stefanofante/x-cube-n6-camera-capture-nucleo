cmake_minimum_required(VERSION 3.20)
project(camera_local_tests LANGUAGES C CXX)
enable_testing()
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
option(ENABLE_SANITIZERS "Run host tests with address/undefined-behavior sanitizers" ON)
if(ENABLE_SANITIZERS AND CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  add_link_options(-fsanitize=address,undefined -no-pie)
endif()
add_compile_options(-Wall -Wextra -Werror)
add_executable(storage_tests storage_tests.cpp)
target_include_directories(storage_tests PRIVATE "${REPO_ROOT}/Inc" "${REPO_ROOT}/Lib/uvcl/Inc")
add_test(NAME static_storage_and_packets COMMAND storage_tests)
add_executable(boot_tests boot_tests.c "${REPO_ROOT}/Tools/FSBL/Src/boot_image.c")
target_include_directories(boot_tests PRIVATE "${REPO_ROOT}/Tools/FSBL/Inc" "${REPO_ROOT}/Inc")
add_test(NAME boot_image_validation COMMAND boot_tests)
add_executable(uvcl_tests uvcl_tests.c "${REPO_ROOT}/Lib/uvcl/Src/uvcl.c")
target_include_directories(uvcl_tests PRIVATE stubs "${REPO_ROOT}/Lib/uvcl/Inc" "${REPO_ROOT}/Lib/uvcl/Src")
# Only hardware-independent paths are linked. Unused firmware initialization
# code still compiles against the HAL declarations but is discarded at link.
target_compile_options(uvcl_tests PRIVATE -ffunction-sections -fdata-sections -Wno-unused-parameter)
target_link_options(uvcl_tests PRIVATE -Wl,--gc-sections)
add_test(NAME uvcl_ownership_and_abort COMMAND uvcl_tests)
